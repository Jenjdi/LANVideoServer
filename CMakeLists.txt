cmake_minimum_required(VERSION 3.20)
project(LANVideoServer VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LOCAL_DEPS "/usr" CACHE PATH "Dependency installation prefix")
option(LANVIDEO_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
foreach(dep sqlite3 cpp-httplib sodium torrent-rasterbar)
  string(REPLACE "-" "_" key ${dep})
  find_library(${key}_LIB NAMES ${dep} HINTS "${LOCAL_DEPS}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" REQUIRED)
endforeach()
add_executable(lanvideo src/main.cpp src/http_server.cpp src/core.cpp src/auth.cpp src/files.cpp src/media.cpp src/uploads.cpp src/bt.cpp src/bt_stream.cpp src/proxy.cpp src/remote.cpp src/shares.cpp src/monitor.cpp)
target_include_directories(lanvideo PRIVATE include)
target_include_directories(lanvideo SYSTEM PRIVATE "${LOCAL_DEPS}/include")
target_compile_definitions(lanvideo PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT CPPHTTPLIB_ZLIB_SUPPORT CPPHTTPLIB_BROTLI_SUPPORT LANVIDEO_WEB_DIR="${CMAKE_SOURCE_DIR}/web" LANVIDEO_VERSION="${PROJECT_VERSION}")
target_compile_options(lanvideo PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(lanvideo PRIVATE ${sqlite3_LIB} ${cpp_httplib_LIB} ${sodium_LIB} ${torrent_rasterbar_LIB} Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
set_target_properties(lanvideo PROPERTIES BUILD_RPATH "${LOCAL_DEPS}/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
target_compile_definitions(lanvideo PRIVATE TORRENT_LINKING_SHARED BOOST_ASIO_ENABLE_CANCELIO BOOST_ASIO_NO_DEPRECATED TORRENT_USE_OPENSSL TORRENT_USE_LIBCRYPTO TORRENT_SSL_PEERS OPENSSL_NO_SSL2)
install(TARGETS lanvideo RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY web/ DESTINATION share/lanvideo/web)
install(DIRECTORY docs/ DESTINATION share/doc/lanvideo/docs FILES_MATCHING PATTERN "*.md" PATTERN "*.json" PATTERN "*.txt")
install(DIRECTORY deploy/ DESTINATION share/doc/lanvideo/deploy FILES_MATCHING PATTERN "*.md" PATTERN "*.example")
install(FILES LICENSE README.md .env.example DESTINATION share/doc/lanvideo)
install(PROGRAMS scripts/auth-client.py scripts/configure-remote.py DESTINATION share/lanvideo/tools)
if(LANVIDEO_SANITIZERS)
  target_compile_options(lanvideo PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lanvideo PRIVATE -fsanitize=address,undefined)
endif()

if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(bt_fixture tests/bt_fixture.cpp)
  target_include_directories(bt_fixture SYSTEM PRIVATE "${LOCAL_DEPS}/include")
  target_compile_definitions(bt_fixture PRIVATE TORRENT_LINKING_SHARED BOOST_ASIO_ENABLE_CANCELIO BOOST_ASIO_NO_DEPRECATED TORRENT_USE_OPENSSL TORRENT_USE_LIBCRYPTO TORRENT_SSL_PEERS OPENSSL_NO_SSL2)
  target_link_libraries(bt_fixture PRIVATE ${torrent_rasterbar_LIB} Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
  set_target_properties(bt_fixture PROPERTIES BUILD_RPATH "${LOCAL_DEPS}/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
  foreach(test integration auth_integration media_integration browser_integration uploads_integration
               bt_integration browser_bt bt_stream_integration browser_bt_stream shares_integration
               remote_integration browser_remote monitor_integration delivery_integration security_regression)
    set(arguments $<TARGET_FILE:lanvideo>)
    if(test MATCHES "bt")
      list(APPEND arguments $<TARGET_FILE:bt_fixture>)
    endif()
    add_test(NAME ${test} COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/${test}.py ${arguments})
    set_tests_properties(${test} PROPERTIES TIMEOUT 180)
  endforeach()
  set_tests_properties(delivery_integration PROPERTIES ENVIRONMENT "CMAKE=${CMAKE_COMMAND}")
endif()
